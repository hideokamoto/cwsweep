import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid>
	<Card title="Inventory the whole Organization" icon="magnifier">
		Starting from the management account's credentials, cwsweep assumes a role into every member account and aggregates log groups in the regions you specify.
	</Card>
	<Card title="Dry-run by default" icon="approve-check">
		Unless you explicitly pass `--execute`, `delete-log-group` and `put-retention-policy` are never called.
	</Card>
	<Card title="Double identity verification" icon="seti:lock">
		The account ID is verified with `sts:get-caller-identity` twice: at scan time and again right before execution.
	</Card>
	<Card title="Audit log is always written" icon="document">
		Every deletion and retention change is recorded in a JSON Lines audit log, and any operation whose log entry cannot be written is aborted.
	</Card>
</CardGrid>
